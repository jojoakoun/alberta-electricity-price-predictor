.DEFAULT_GOAL := help

PYTHON ?= .venv/bin/python
PYTEST ?= .venv/bin/pytest
MAIN_MAKEFILE ?= Makefile

.PHONY: \
	help \
	verify

help:
	@echo ""
	@echo "WattWise — validated daily commands"
	@echo ""
	@echo "  make -f Makefile.useful verify"
	@echo "      Validate configuration, Python, API, frontend, lint, and build."
	@echo ""

verify:
	$(MAKE) -f $(MAIN_MAKEFILE) verify \
		PYTHON=$(PYTHON) \
		PYTEST=$(PYTEST)
