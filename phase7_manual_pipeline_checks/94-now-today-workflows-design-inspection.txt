0001: const { pool } = require("../db/pool");
0002: 
0003: const MARKET_CONTEXT_LOOKBACK_HOURS = 24 * 30;
0004: 
0005: // Forecast-only rows are excluded because Now must describe an observed price.
0006: async function getLatestFinalizedPrice() {
0007:   const { rows } = await pool.query(`
0008:     SELECT
0009:       datetime_utc,
0010:       actual_price
0011:     FROM hourly_prices
0012:     WHERE actual_price IS NOT NULL
0013:     ORDER BY datetime_utc DESC
0014:     LIMIT 1;
0015:   `);
0016: 
0017:   return rows[0] ?? null;
0018: }
0019: 
0020: // Bound the query to the documented 30-day distribution used for market
0021: // context; the outer order keeps the repository result chronological.
0022: async function getRecentFinalizedPrices(
0023:   limit = MARKET_CONTEXT_LOOKBACK_HOURS,
0024: ) {
0025:   const { rows } = await pool.query(
0026:     `
0027:       SELECT actual_price
0028:       FROM (
0029:         SELECT
0030:           datetime_utc,
0031:           actual_price
0032:         FROM hourly_prices
0033:         WHERE actual_price IS NOT NULL
0034:         ORDER BY datetime_utc DESC
0035:         LIMIT $1
0036:       ) AS recent
0037:       ORDER BY datetime_utc ASC;
0038:     `,
0039:     [limit],
0040:   );
0041: 
0042:   return rows;
0043: }

0001: """Incrementally synchronize operational AESO prices with PostgreSQL."""
0002: 
0003: from datetime import datetime
0004: from zoneinfo import ZoneInfo
0005: 
0006: import pandas as pd
0007: 
0008: from electricity_predictor.data.aeso_api import (
0009:   fetch_pool_price_report,
0010:   normalize_pool_price_report,
0011: )
0012: from electricity_predictor.worker.persistence import (
0013:   get_latest_hourly_price_timestamp,
0014:   upsert_hourly_prices,
0015: )
0016: 
0017: 
0018: ALBERTA_TIMEZONE = ZoneInfo("America/Edmonton")
0019: OPERATIONAL_REFRESH_OVERLAP_DAYS = 2
0020: 
0021: 
0022: def derive_api_start_date(
0023:   latest_database_timestamp: datetime | pd.Timestamp,
0024: ) -> str:
0025:   """Return a bounded Alberta start date derived from PostgreSQL state.
0026: 
0027:   The overlap captures recent actual-price finalizations and forecast revisions
0028:   without downloading or rewriting the full historical dataset.
0029:   """
0030:   latest_timestamp = pd.Timestamp(
0031:     latest_database_timestamp
0032:   )
0033: 
0034:   if pd.isna(latest_timestamp):
0035:     raise ValueError(
0036:       "Latest database timestamp must not be missing."
0037:     )
0038: 
0039:   if latest_timestamp.tzinfo is None:
0040:     latest_timestamp = latest_timestamp.tz_localize(
0041:       "UTC"
0042:     )
0043:   else:
0044:     latest_timestamp = latest_timestamp.tz_convert(
0045:       "UTC"
0046:     )
0047: 
0048:   latest_alberta_date = latest_timestamp.tz_convert(
0049:     ALBERTA_TIMEZONE
0050:   ).normalize()
0051:   overlap_start = latest_alberta_date - pd.Timedelta(
0052:     days=OPERATIONAL_REFRESH_OVERLAP_DAYS
0053:   )
0054: 
0055:   return overlap_start.strftime("%Y-%m-%d")
0056: 
0057: 
0058: def get_current_api_end_date() -> str:
0059:   """Return the current Alberta market date for AESO."""
0060:   return datetime.now(
0061:     ALBERTA_TIMEZONE
0062:   ).strftime("%Y-%m-%d")
0063: 
0064: 
0065: def build_operational_hourly_price_rows(
0066:   api_data: pd.DataFrame,
0067: ) -> pd.DataFrame:
0068:   """Build normalized AESO rows for the operational PostgreSQL schema."""
0069:   operational_data = api_data.copy()
0070: 
0071:   operational_data[
0072:     "datetime_universal_time"
0073:   ] = pd.to_datetime(
0074:     operational_data[
0075:       "datetime_universal_time"
0076:     ],
0077:     utc=True,
0078:   )
0079: 
0080:   # This AESO report does not carry actual AIL. Preserve any database value
0081:   # during the conflict update instead of manufacturing one.
0082:   operational_data[
0083:     "alberta_internal_load"
0084:   ] = pd.NA
0085: 
0086:   return operational_data[
0087:     [
0088:       "datetime_universal_time",
0089:       "actual_price",
0090:       "forecast_price",
0091:       "alberta_internal_load",
0092:     ]
0093:   ].sort_values(
0094:     "datetime_universal_time"
0095:   ).reset_index(drop=True)
0096: 
0097: 
0098: def synchronize_operational_prices(
0099:   end_date: str | None = None,
0100: ) -> int:
0101:   """Synchronize a bounded AESO overlap into operational PostgreSQL.
0102: 
0103:   PostgreSQL is the recurring worker's source of truth. An empty table is an
0104:   explicit seed prerequisite failure, never a reason to read a deployment CSV.
0105:   """
0106:   latest_database_timestamp = (
0107:     get_latest_hourly_price_timestamp()
0108:   )
0109: 
0110:   if latest_database_timestamp is None:
0111:     raise RuntimeError(
0112:       "PostgreSQL contains no hourly prices; run the one-time "
0113:       "historical database seed before the operational worker."
0114:     )
0115: 
0116:   start_date = derive_api_start_date(
0117:     latest_database_timestamp
0118:   )
0119: 
0120:   if end_date is None:
0121:     end_date = get_current_api_end_date()
0122: 
0123:   api_report = fetch_pool_price_report(
0124:     start_date=start_date,
0125:     end_date=end_date,
0126:   )
0127:   api_data = normalize_pool_price_report(
0128:     api_report
0129:   )
0130:   operational_data = build_operational_hourly_price_rows(
0131:     api_data
0132:   )
0133: 
0134:   return upsert_hourly_prices(
0135:     data=operational_data,
0136:     source="aeso_api",
0137:   )
