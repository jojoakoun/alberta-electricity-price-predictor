0016: function calculateFreshness(
0017:   timestamp,
0018:   now,
0019:   policy,
0020:   timestampName,
0021: ) {
0022:   const timestampDate = new Date(timestamp);
0023:   const currentDate = new Date(now);
0024: 
0025:   if (
0026:     Number.isNaN(timestampDate.getTime()) ||
0027:     Number.isNaN(currentDate.getTime())
0028:   ) {
0029:     throw new TypeError("Freshness requires valid dates.");
0030:   }
0031: 
0032:   const ageMs =
0033:     currentDate.getTime() - timestampDate.getTime();
0034: 
0035:   if (ageMs < 0) {
0036:     throw new RangeError(
0037:       `${timestampName} cannot be in the future.`,
0038:     );
0039:   }
0040: 
0041:   if (ageMs <= policy.highConfidenceMaxAgeMs) {
0042:     return {
0043:       confidence: "high",
0044:       stale: false,
0045:     };
0046:   }
0047: 
0048:   if (ageMs <= policy.moderateConfidenceMaxAgeMs) {
0049:     return {
0050:       confidence: "moderate",
0051:       stale: true,
0052:     };
0053:   }
0054: 
0055:   return {
0056:     confidence: "low",

0001: /** Return a linearly interpolated quantile from sorted finite values. */
0002: function quantile(sortedValues, probability) {
0003:   const position = (
0004:     sortedValues.length - 1
0005:   ) * probability;
0006: 
0007:   const lowerIndex = Math.floor(position);
0008:   const upperIndex = Math.ceil(position);
0009: 
0010:   if (lowerIndex === upperIndex) {
0011:     return sortedValues[lowerIndex];
0012:   }
0013: 
0014:   const weight = position - lowerIndex;
0015: 
0016:   return (
0017:     sortedValues[lowerIndex] * (1 - weight)
0018:     + sortedValues[upperIndex] * weight
0019:   );
0020: }

0022: function parseFinitePrice(value, label) {
0023:   const isNumericValue =
0024:     typeof value === "number"
0025:     || (
0026:       typeof value === "string"
0027:       && value.trim() !== ""

0096: function buildMarketThresholds(
0097:   currentPrice,
0098:   recentPriceRows,
0099: ) {
0100:   const {

0114:     recentPrices,

0119:     thirdQuartile - firstQuartile

0123:     price,
0124:     firstQuartile,
0125:     thirdQuartile,

0127:       thirdQuartile

0001: """Fetch and normalize AESO prices for research and operational refreshes."""
0002: 
0003: import os
0004: from typing import Any
0005: 
0006: import pandas as pd
0007: import requests
0008: from dotenv import load_dotenv
0009: 
0010: from electricity_predictor.config import load_configuration
0011: 
0012: 
0013: API_COLUMNS = {
0014:   "begin_datetime_utc": "datetime_universal_time",
0015:   "begin_datetime_mpt": "datetime_local_time",
0016:   "pool_price": "actual_price",
0017:   "forecast_pool_price": "forecast_price",
0018: }
0019: 
0020: 
0021: def fetch_pool_price_report(start_date: str, end_date: str) -> dict[str, Any]:
0022:   """Fetch an inclusive Alberta market-date range from the AESO API."""
0023:   load_dotenv()
0024: 
0025:   configuration = load_configuration()
0026:   base_url = os.getenv(
0027:     "AESO_API_BASE_URL",
0028:     configuration["api"]["aeso_base_url"],
0029:   )
0030:   subscription_key = os.getenv("AESO_API_SUBSCRIPTION_KEY")
0031: 
0032:   if not subscription_key:
0033:     raise ValueError("Missing AESO_API_SUBSCRIPTION_KEY in local environment.")
0034: 
0035:   url = f"{base_url}{configuration['api']['aeso_pool_price_endpoint']}"
0036:   params = {
0037:     "startDate": start_date,
0038:     "endDate": end_date,
0039:   }
0040:   headers = {
0041:     "API-KEY": subscription_key,
0042:   }
0043: 
0044:   response = requests.get(
0045:     url,
0046:     headers=headers,
0047:     params=params,
0048:     timeout=30,
0049:   )
0050:   response.raise_for_status()
0051: 
0052:   return response.json()
0053: 
0054: 
0055: def normalize_pool_price_report(report: dict[str, Any]) -> pd.DataFrame:
0056:   """Normalize AESO records into the shared hourly price schema.
0057: 
0058:   Recent actual prices may be null until AESO finalizes the source hour, but
0059:   every row must include its forecast and both market timestamps.
0060:   """
0061:   records = report.get("return", {}).get("Pool Price Report", [])
0062: 
0063:   if not records:
0064:     raise ValueError("AESO response does not contain pool price records.")
0065: 
0066:   data = pd.DataFrame(records)
0067:   data = data.rename(columns=API_COLUMNS)
0068:   data = data[list(API_COLUMNS.values())]
0069: 
0070:   data["datetime_universal_time"] = pd.to_datetime(
0071:     data["datetime_universal_time"]
0072:   )
0073:   data["datetime_local_time"] = pd.to_datetime(
0074:     data["datetime_local_time"]
0075:   )
0076:   data["actual_price"] = pd.to_numeric(data["actual_price"])
0077:   data["forecast_price"] = pd.to_numeric(data["forecast_price"])
0078: 
0079:   data = data.sort_values(
0080:     "datetime_universal_time"
0081:   ).reset_index(drop=True)
0082:   validate_pool_price_data(data)
0083: 
0084:   return data
0085: 
0086: 
0087: def validate_pool_price_data(data: pd.DataFrame) -> None:
0088:   """Validate normalized AESO timestamps and required forecast values."""
0089:   required_columns = set(API_COLUMNS.values())
0090:   missing_columns = required_columns - set(data.columns)
0091: 
0092:   if missing_columns:
0093:     raise ValueError(f"Missing required API columns: {sorted(missing_columns)}")
0094: 
0095:   # Actual prices may remain null while an hour is being finalized. These
0096:   # fields are required to preserve a usable forecast-only operational row.
0097:   required_non_null_columns = [
0098:     "datetime_universal_time",
0099:     "datetime_local_time",
0100:     "forecast_price",
0101:   ]
0102: 
0103:   for column in required_non_null_columns:
0104:     if data[column].isna().any():
0105:       raise ValueError(f"API column contains missing values: {column}")
0106: 
0107:   if data["datetime_universal_time"].duplicated().any():
0108:     raise ValueError("API data contains duplicate UTC timestamps.")
0109: 
0110:   if not data["datetime_universal_time"].is_monotonic_increasing:
0111:     raise ValueError("API data must be sorted by datetime_universal_time.")
0112: 
0113: 
0114: if __name__ == "__main__":
0115:   report = fetch_pool_price_report(
0116:     start_date="2026-06-01",
0117:     end_date="2026-06-04",
0118:   )
0119:   data = normalize_pool_price_report(report)
0120: 
0121:   print(data.head())
0122:   print(data.shape)
0123:   print(data.columns.tolist())
